#include <Arduino.h>
#include <Wire.h>
#include <Adafruit_SHT31.h>
#include <LiquidCrystal_I2C.h>   // ← original, no patch needed now

#define SHT_SDA  18
#define SHT_SCL  22
#define LCD_SDA  21
#define LCD_SCL  23

Adafruit_SHT31    sht30;
LiquidCrystal_I2C lcd(0x27, 16, 2);

void setup() {
    Serial.begin(115200);
    delay(1000);
    Serial.println("=== Dual I2C Bus Test ===");

    // SHT30 on Wire (GPIO 18/22)
    Wire.begin(SHT_SDA, SHT_SCL, 100000);
    if (!sht30.begin(0x44)) {
        Serial.println("[ERROR] SHT30 not found!");
    } else {
        Serial.println("[SHT30] OK");
    }

    // LCD on Wire1 (GPIO 21/23) — library now uses Wire1 internally
    Wire1.begin(LCD_SDA, LCD_SCL, 100000);
    lcd.init();
    lcd.backlight();
    lcd.setCursor(0, 0);
    lcd.print("Dual I2C Test");
    lcd.setCursor(0, 1);
    lcd.print("Starting...");
    Serial.println("[LCD] OK");
    delay(2000);
}

void loop() {
    float t = sht30.readTemperature();
    float h = sht30.readHumidity();

    if (!isnan(t) && !isnan(h)) {
        Serial.printf("[Sensor] T=%.1fC  H=%.1f%%RH\n", t, h);

        lcd.setCursor(0, 0);
        lcd.print("Temp: ");
        lcd.print(t, 1);
        lcd.print("C   ");

        lcd.setCursor(0, 1);
        lcd.print("Hum:  ");
        lcd.print(h, 1);
        lcd.print("% RH  ");

    } else {
        Serial.println("[ERROR] SHT30 Read Failed!");
        lcd.setCursor(0, 1);
        lcd.print("Sensor Error!   ");
    }
    delay(2000);
}
