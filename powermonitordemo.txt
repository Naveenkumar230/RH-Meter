function pad(n){ return n < 10 ? '0'+n : ''+n; }

function todayIST(){
  const now = new Date(), ist = new Date(now.getTime() + 5.5*3600000);
  return ist.toISOString().slice(0,10);
}

function randVrms(base=228){ return parseFloat((base + (Math.random()*8 - 4)).toFixed(1)); }
function randLow(){          return parseFloat((1 + Math.random()*3).toFixed(1)); }

function isDeviceOnAt(deviceId, hourIST){
  const p = DEVICE_PROFILES[deviceId];
  if(p.offlineDevice) return false;
  for(const [s,e] of p.offHours){
    if(hourIST >= s && hourIST < e) return false;
  }
  return true;
}

// ── Generate today's 10-min interval records ──
function generateTodayHistory(deviceId){
  const p = DEVICE_PROFILES[deviceId];
  const records = [];
  const nowIST  = new Date(Date.now() + 5.5*3600*1000);
  const baseIST = new Date(nowIST);
  baseIST.setUTCHours(0,0,0,0);
  // subtract 5.5h offset to get UTC midnight of IST today
  const startUTC = new Date(baseIST.getTime() - 5.5*3600*1000);

  const currentMinutes = nowIST.getUTCHours()*60 + nowIST.getUTCMinutes();

  for(let m = 0; m < currentMinutes; m += 10){
    const ts   = new Date(startUTC.getTime() + m*60000 + 5.5*3600000); // IST timestamp
    const utcT = new Date(startUTC.getTime() + m*60000);
    const hIST = Math.floor(m/60);
    const isOn = isDeviceOnAt(deviceId, hIST);

    records.push({
      timestamp: utcT.toISOString(),
      vrms:      isOn ? randVrms() : randLow(),
      online:    isOn,
    });
  }
  return records;
}

// ── Generate history for a past date (full 24h) ──
function generateDayHistory(deviceId, dateStr){
  // dateStr = 'YYYY-MM-DD' in IST
  const records = [];
  // Build UTC start of IST day
  const [y,mo,d] = dateStr.split('-').map(Number);
  const istMidnight = new Date(Date.UTC(y, mo-1, d, 0, 0, 0) - 5.5*3600*1000);

  for(let m = 0; m < 24*60; m += 10){
    const utcT = new Date(istMidnight.getTime() + m*60000);
    const hIST = Math.floor(m/60);
    const isOn = isDeviceOnAt(deviceId, hIST);

    // Add occasional random glitches for realism (10% chance per interval for a brief flip)
    const glitch = !DEVICE_PROFILES[deviceId].offlineDevice && Math.random() < 0.05;
    const actualOn = glitch ? !isOn : isOn;

    records.push({
      timestamp: utcT.toISOString(),
      vrms:      actualOn ? randVrms() : randLow(),
      online:    actualOn,
    });
  }
  return records;
}

// ── Get date strings for last 30 days in IST ──
function getLast30Days(){
  const days = [];
  const now  = new Date(Date.now() + 5.5*3600*1000);
  for(let i = 29; i >= 0; i--){
    const d = new Date(now);
    d.setUTCDate(d.getUTCDate() - i);
    days.push(d.toISOString().slice(0,10));
  }
  return days;
}

  DEHUM_DEVICES.forEach(dev => {
    const p      = DEVICE_PROFILES[dev.id];
    const nowIST = new Date(Date.now() + 5.5*3600*1000);
    const hIST   = nowIST.getUTCHours();
    const online = isDeviceOnAt(dev.id, hIST);
    deviceCache[dev.id] = {
      vrms:     online ? randVrms() : randLow(),
      online,
      lastSeen: new Date().toISOString(),
    };
  });
}

// ── Seed today's history for all devices ──
function seedTodayHistory(){
  DEHUM_DEVICES.forEach(dev => {
    deviceHistoryCache[dev.id+':today'] = generateTodayHistory(dev.id);
  });
}

// ── Get history for date range (fake API) ──
function getFakeHistory(deviceId, from, to){
  const fromD = new Date(from+'T00:00:00.000+05:30');
  const toD   = new Date(to  +'T23:59:59.999+05:30');
  const today = todayIST();
  const days  = getLast30Days();
  const allRecords = [];

  days.forEach(day => {
    const dayStart = new Date(day+'T00:00:00.000+05:30');
    const dayEnd   = new Date(day+'T23:59:59.999+05:30');
    if(dayEnd < fromD || dayStart > toD) return;

    if(day === today){
      const cached = deviceHistoryCache[deviceId+':today'] || generateTodayHistory(deviceId);
      cached.forEach(r => {
        const rt = new Date(r.timestamp);
        if(rt >= fromD && rt <= toD) allRecords.push(r);
      });
    } else {
      const cacheKey = deviceId+':'+day;
      if(!deviceHistoryCache[cacheKey]){
        deviceHistoryCache[cacheKey] = generateDayHistory(deviceId, day);
      }
      deviceHistoryCache[cacheKey].forEach(r => {
        const rt = new Date(r.timestamp);
        if(rt >= fromD && rt <= toD) allRecords.push(r);
      });
    }
  });

  return allRecords.sort((a,b) => new Date(a.timestamp) - new Date(b.timestamp));
}

//  UI — SPARKLINE  (per-hour ON bar on card)
// ═══════════════════════════════════════════════════════════════
function buildSparkline(canvasId, deviceId, data){
  const canvas = document.getElementById(canvasId);
  if(!canvas) return;
  if(sparklineCharts[deviceId]){ sparklineCharts[deviceId].destroy(); }

  const onMins  = new Array(24).fill(0);
  const offMins = new Array(24).fill(0);
  const labels  = Array.from({length:24}, (_,h) => pad(h)+':00');

  data.forEach(r => {
    const d   = new Date(r.timestamp);
    const ist = new Date(d.getTime() + 5.5*3600000);
    const h   = ist.getUTCHours();
    const isOn = !!(r.online || parseFloat(r.vrms||0) >= VOLTAGE_THRESHOLD);
    if(isOn) onMins[h]  += 10;
    else     offMins[h] += 10;
  });

  const colors = onMins.map((on,h) => {
    if(onMins[h]+offMins[h] === 0) return 'transparent';
    return on > 0 ? 'rgba(22,163,74,0.75)' : 'rgba(220,38,38,0.5)';
  });

  sparklineCharts[deviceId] = new Chart(canvas.getContext('2d'), {
    type: 'bar',
    data: { labels, datasets:[{ data:onMins.map((v,h)=>onMins[h]+offMins[h]>0?Math.max(v,1):null), backgroundColor:colors, borderRadius:2, borderSkipped:false }] },
    options: {
      responsive:true, maintainAspectRatio:false, animation:{duration:200},
      plugins:{ legend:{display:false}, tooltip:{enabled:false} },
      scales:{ x:{display:false}, y:{display:false, min:0, max:60} }
    }
  });
}

        <div class="dehum-voltage-unit">V RMS</div>
      </div>
      <canvas class="dehum-sparkline" id="${sparkId}" height="44"></canvas>
      <div class="dehum-card-footer">
        <span class="dehum-last-seen">🕐 ${lastStr}</span>

//  CHART — 24-HOUR TIMELINE
//  Y-axis: 0–60 minutes | X-axis: 00:00–23:00
//  Green bars = ON minutes per hour | Red bars = OFF minutes per hour
// ═══════════════════════════════════════════════════════════════
function renderTimelineChart(hist){
  if(chartTimeline){ chartTimeline.destroy(); chartTimeline=null; }

  const labels  = Array.from({length:24}, (_,h) => pad(h)+':00');
  const onMins  = new Array(24).fill(0);
  const offMins = new Array(24).fill(0);

  hist.forEach(r => {
    const d   = new Date(r.timestamp);
    const ist = new Date(d.getTime() + 5.5*3600000);
    const h   = ist.getUTCHours();
    const isOn = !!(r.online || parseFloat(r.vrms||0) >= VOLTAGE_THRESHOLD);
    if(isOn) onMins[h]  += 10;
    else     offMins[h] += 10;
  });
  onMins.forEach((v,i)  => { onMins[i]  = Math.min(v,60); });
  offMins.forEach((v,i) => { offMins[i] = Math.min(v,60); });

  const ctx = document.getElementById('chartTimeline').getContext('2d');
  chartTimeline = new Chart(ctx, {
    type: 'bar',
    data: {
      labels,
      datasets:[
        { label:'ON',  data:onMins,  backgroundColor:'rgba(22,163,74,0.75)', borderRadius:3, borderSkipped:false },
        { label:'OFF', data:offMins, backgroundColor:'rgba(220,38,38,0.6)',  borderRadius:3, borderSkipped:false }
      ]
    },
    options:{
      responsive:true, maintainAspectRatio:false, animation:{duration:400},
      plugins:{
        legend:{display:false},
        tooltip:{
          callbacks:{
            title: ctx => `Hour: ${ctx[0].label}`,
            label: ctx => ` ${ctx.dataset.label}: ${ctx.parsed.y} min`,
            afterBody: items => {
              const h   = items[0].dataIndex;
              const on  = onMins[h], off = offMins[h];
              if(on+off===0) return ['No data recorded'];
              return [`Uptime: ${Math.round((on/(on+off))*100)}%`];
            }
          }
        }
      },
      scales:{
        x:{ ticks:{color:'#94a3b8', autoSkip:false, maxRotation:45, callback:(v,i)=>i%2===0?labels[i]:''}, grid:{color:'rgba(15,23,42,0.05)'} },
        y:{ min:0, max:60, ticks:{color:'#94a3b8', stepSize:10, callback:v=>`${v} min`}, grid:{color:'rgba(15,23,42,0.05)'}, title:{display:true,text:'Minutes per hour',color:'#94a3b8',font:{size:11}} }
      }
    }
  });
}
